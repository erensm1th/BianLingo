cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
project(BianLingoCT2 LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  # Keep Abseil feature probes on the same dialect as SentencePiece, including
  # nested projects whose older CMake policies omit the dialect in try_compile.
  string(APPEND CMAKE_REQUIRED_FLAGS " /std:c++17")
endif()
unset(ABSL_INTERNAL_AT_LEAST_CXX17 CACHE)
unset(ABSL_INTERNAL_AT_LEAST_CXX20 CACHE)
foreach(pair BUILD_SHARED_LIBS BUILD_CLI BUILD_TESTS WITH_MKL WITH_DNNL WITH_CUDA WITH_CUDNN WITH_HIP WITH_OPENBLAS)
  set(${pair} OFF CACHE BOOL "" FORCE)
endforeach()
set(WITH_RUY ON CACHE BOOL "" FORCE)
set(OPENMP_RUNTIME NONE CACHE STRING "" FORCE)
set(RUY_MINIMAL_BUILD ON CACHE BOOL "" FORCE)
set(CPUINFO_BUILD_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(CPUINFO_BUILD_MOCK_TESTS OFF CACHE BOOL "" FORCE)
set(CPUINFO_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
add_subdirectory("${CT2_SOURCE}" ctranslate2)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_MSVC_MT_BUILD ON CACHE BOOL "" FORCE)
set(ABSL_MSVC_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
add_subdirectory("${SPM_SOURCE}" sentencepiece)
add_executable(BianLingoCT2 main.cpp)
target_compile_features(BianLingoCT2 PRIVATE cxx_std_17)
target_include_directories(BianLingoCT2 PRIVATE "${CT2_SOURCE}/include" "${SPM_SOURCE}" "${SPM_SOURCE}/src")
target_link_libraries(BianLingoCT2 PRIVATE ctranslate2 sentencepiece-static)
if(MSVC)
  target_compile_options(BianLingoCT2 PRIVATE /utf-8 /W4 /EHsc)
  target_compile_definitions(BianLingoCT2 PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_options(BianLingoCT2 PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/windows.manifest)
endif()
