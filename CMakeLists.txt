if(WIN32)
  find_package(Tesseract CONFIG REQUIRED)
  find_package(zstd CONFIG REQUIRED)
  # FindLibArchive can choose vcpkg's debug archive for all configurations.
  # Supply both static CRT variants explicitly to avoid /MT vs /MTd mixing.
  if(TARGET LibArchive::LibArchive AND VCPKG_TARGET_TRIPLET MATCHES "windows-static")
    set(_archive_prefix "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}")
    if(EXISTS "${_archive_prefix}/lib/archive.lib")
      set_target_properties(LibArchive::LibArchive PROPERTIES
        IMPORTED_CONFIGURATIONS "DEBUG;RELEASE"
        IMPORTED_LOCATION_RELEASE "${_archive_prefix}/lib/archive.lib"
        IMPORTED_LOCATION_DEBUG "${_archive_prefix}/debug/lib/archive.lib"
        MAP_IMPORTED_CONFIG_RELWITHDEBINFO Release MAP_IMPORTED_CONFIG_MINSIZEREL Release)
    endif()
  endif()
  find_package(LibArchive REQUIRED)
  target_link_libraries(bianlingo_core PUBLIC Tesseract::libtesseract zstd::libzstd LibArchive::LibArchive d3d11 dxgi windowsapp dwmapi)
  target_sources(bianlingo_core PRIVATE src/platform/windows_ocr.cpp src/platform/windows_bergamot.cpp
    src/platform/windows_ctranslate2.cpp src/platform/windows_local_models.cpp src/platform/windows_ct2_models.cpp src/platform/windows_runtime_probe.cpp src/platform/windows_ollama.cpp)
  target_sources(bianlingo_core PRIVATE src/platform/windows.cpp src/platform/windows_package.cpp src/download_windows.cpp)
  set(BL_PACKAGE_PLATFORM "windows-x64")
  target_link_libraries(bianlingo_core PUBLIC crypt32 winhttp ws2_32 iphlpapi advapi32 user32 gdi32 shell32)
  enable_language(RC)
  add_executable(BianLingo WIN32 src/ui/windows.cpp src/ui/windows_widgets.cpp resources/windows.rc)
  target_link_libraries(BianLingo PRIVATE bianlingo_core comctl32 comdlg32 ole32 dwmapi imm32 gdiplus)
  target_include_directories(BianLingo PRIVATE resources "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_options(BianLingo PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/windows.manifest)
elseif(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 20)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  target_sources(bianlingo_core PRIVATE src/platform/macos.mm src/download_macos.mm)
  if(CMAKE_OSX_ARCHITECTURES STREQUAL "x86_64" OR CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
    set(BL_PACKAGE_PLATFORM "macos-x64")
  else()
    set(BL_PACKAGE_PLATFORM "macos-arm64")
  endif()
  target_compile_options(bianlingo_core PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
  target_link_libraries(bianlingo_core PUBLIC "-framework AppKit" "-framework Security" "-framework IOKit" "-framework ScreenCaptureKit" "-framework CoreMedia" "-framework CoreVideo" "-framework Carbon")
  add_executable(BianLingo MACOSX_BUNDLE src/ui/macos.mm)
  target_include_directories(BianLingo PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_options(BianLingo PRIVATE -fobjc-arc)
  target_link_libraries(BianLingo PRIVATE bianlingo_core "-framework QuartzCore")
  set_target_properties(BianLingo PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/resources/Info.plist.in")
else()
  message(FATAL_ERROR "The desktop adapters currently target Windows and macOS only")
endif()
